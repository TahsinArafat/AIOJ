# AIOJ robots.txt
#
# Allow crawling of the public surfaces and point crawlers at the sitemap.
# Authenticated areas are disallowed: they redirect to login for anonymous
# requests, so letting them be crawled only wastes crawl budget and produces
# soft-404s in search console.

User-agent: *
Allow: /
Allow: /problems
Allow: /contests
Allow: /blog
Allow: /rankings

# Login-gated or user-specific surfaces: nothing useful to index.
Disallow: /admin
Disallow: /settings
Disallow: /submissions
Disallow: /ide
Disallow: /setter
Disallow: /api/
Disallow: /login
Disallow: /register
Disallow: /forgot-password

# Query-string result pages are near-duplicates of the listing pages.
Disallow: /*?*sort=
Disallow: /*?*page=

# The origin is deployment configuration, so this is filled in at build time
# from PUBLIC_ORIGIN by `scripts/prerender-robots.mjs` — the same value
# index.html's og:image and the legal pages' canonical tags use. A relative
# Sitemap URL is invalid per the robots.txt spec, so the line is dropped
# entirely when the origin is unset rather than shipped as a broken path.
Sitemap: %SITE_ORIGIN%/sitemap.xml
