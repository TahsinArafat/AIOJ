# AIOJ — Online Judge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a lightweight online judge with Go backend, React frontend, go-judge sandbox, contest management, and vJudge bot support.

**Architecture:** Unified Go monolith with clean module boundaries. Single binary (API + judge worker) communicates with PostgreSQL and a go-judge sidecar. Internal channel-based queue for submissions — swappable to Redis. React SPA frontend via REST + WebSocket.

**Tech Stack:** Go 1.26, Chi router, PostgreSQL 18, golang-migrate, criyle/go-judge, JWT auth, gorilla/websocket, React 19 + Vite + TypeScript + Tailwind CSS. Docker Compose deployment.

```go
package config

import (
    "os"
    "gopkg.in/yaml.v3"
)

type Config struct {
    Server   ServerConfig   `yaml:"server"`
    Database DatabaseConfig `yaml:"database"`
    Auth     AuthConfig     `yaml:"auth"`
    Judge    JudgeConfig    `yaml:"judge"`
    LangDir  string         `yaml:"lang_dir"`
}

type ServerConfig struct {
    Host string `yaml:"host"`
    Port int    `yaml:"port"`
}

type DatabaseConfig struct {
    Host     string `yaml:"host"`
    Port     int    `yaml:"port"`
    User     string `yaml:"user"`
    Password string `yaml:"password"`
    Name     string `yaml:"name"`
    MaxOpen  int    `yaml:"max_open"`
    MaxIdle  int    `yaml:"max_idle"`
}

type AuthConfig struct {
    JWTSecret  string `yaml:"jwt_secret"`
    AccessTTL  string `yaml:"access_ttl"`
    RefreshTTL string `yaml:"refresh_ttl"`
}

type JudgeConfig struct {
    Endpoint    string `yaml:"endpoint"`
    Concurrency int    `yaml:"concurrency"`
    MaxCodeSize int64  `yaml:"max_code_size"`
}

func Load(path string) (*Config, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        return nil, err
    }
    var cfg Config
    if err := yaml.Unmarshal(data, &cfg); err != nil {
        return nil, err
    }
    if v := os.Getenv("DB_HOST"); v != "" { cfg.Database.Host = v }
    if v := os.Getenv("DB_PASSWORD"); v != "" { cfg.Database.Password = v }
    if v := os.Getenv("JUDGE_ENDPOINT"); v != "" { cfg.Judge.Endpoint = v }
    if v := os.Getenv("JWT_SECRET"); v != "" { cfg.Auth.JWTSecret = v }
    return &cfg, nil
}
```

```go
// internal/store/postgres/conn.go
package postgres

import (
    "database/sql"
    "fmt"
    "time"
    _ "github.com/lib/pq"
    "github.com/tahsinarafat/aioj/internal/config"
)

func Connect(cfg config.DatabaseConfig) (*sql.DB, error) {
    dsn := fmt.Sprintf("host=%s port=%d user=%s password=%s dbname=%s sslmode=disable",
        cfg.Host, cfg.Port, cfg.User, cfg.Password, cfg.Name)
    db, err := sql.Open("postgres", dsn)
    if err != nil { return nil, fmt.Errorf("sql open: %w", err) }
    db.SetMaxOpenConns(cfg.MaxOpen)
    db.SetMaxIdleConns(cfg.MaxIdle)
    db.SetConnMaxLifetime(5 * time.Minute)
    if err := db.Ping(); err != nil { return nil, fmt.Errorf("db ping: %w", err) }
    return db, nil
}
```

- [ ] **Step 4: Write migration runner**

```go
// cmd/migrate/main.go
package main

import (
    "flag"
    "log"
    "os"
    "github.com/golang-migrate/migrate/v4"
    _ "github.com/golang-migrate/migrate/v4/database/postgres"
    _ "github.com/golang-migrate/migrate/v4/source/file"
)

func main() {
    dsn := os.Getenv("DB_DSN")
    if dsn == "" {
        dsn = "postgres://aioj:aioj_dev@localhost:5432/aioj?sslmode=disable"
    }
    dir := flag.String("dir", "internal/store/migrations", "migrations directory")
    flag.Parse()
    cmd := flag.Arg(0)
    m, err := migrate.New("file://"+*dir, dsn)
    if err != nil { log.Fatalf("migrate new: %v", err) }
    switch cmd {
    case "up":
        if err := m.Up(); err != nil && err != migrate.ErrNoChange { log.Fatalf("migrate up: %v", err) }
        log.Println("migration up complete")
    case "down":
        if err := m.Down(); err != nil && err != migrate.ErrNoChange { log.Fatalf("migrate down: %v", err) }
        log.Println("migration down complete")
    default:
        log.Fatalf("unknown command: %s (use up or down)", cmd)
    }
}
```

```go
const bcryptCost = 12

func HashPassword(password string) (string, error) {
    bytes, err := bcrypt.GenerateFromPassword([]byte(password), bcryptCost)
    return string(bytes), err

func CheckPassword(password, hash string) bool {
```

```go
package auth

import (
    "crypto/rand"
    "crypto/sha256"
    "encoding/hex"
    "fmt"
    "time"
    "github.com/golang-jwt/jwt/v5"
)

type Claims struct {
    UserID   string `json:"uid"`
    Username string `json:"uname"`
    Role     string `json:"role"`
    jwt.RegisteredClaims
}

type JWTManager struct {
    secret     []byte
    accessTTL  time.Duration
    refreshTTL time.Duration
}

func NewJWTManager(secret string, accessTTL, refreshTTL time.Duration) *JWTManager {
    return &JWTManager{secret: []byte(secret), accessTTL: accessTTL, refreshTTL: refreshTTL}
}

func (m *JWTManager) GenerateAccessToken(userID, username, role string) (string, error) {
    claims := Claims{userID, username, role, jwt.RegisteredClaims{
        ExpiresAt: jwt.NewNumericDate(time.Now().Add(m.accessTTL)),
        IssuedAt:  jwt.NewNumericDate(time.Now()),
    }}
    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    return token.SignedString(m.secret)
}

func (m *JWTManager) ValidateToken(tokenStr string) (*Claims, error) {
    token, err := jwt.ParseWithClaims(tokenStr, &Claims{}, func(t *jwt.Token) (interface{}, error) {
        if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
            return nil, fmt.Errorf("unexpected signing method: %v", t.Header["alg"])
        }
        return m.secret, nil
    })
    if err != nil { return nil, err }
    claims, ok := token.Claims.(*Claims)
    if !ok || !token.Valid { return nil, fmt.Errorf("invalid token") }
    return claims, nil
}

func (m *JWTManager) GenerateRefreshToken() (raw, hashed string) {
    bytes := make([]byte, 32)
    rand.Read(bytes)
    raw = hex.EncodeToString(bytes)
    h := sha256.Sum256([]byte(raw))
    hashed = hex.EncodeToString(h[:])
    return
}

func (m *JWTManager) RefreshTTL() time.Duration { return m.refreshTTL }
```

```go
const UserContextKey contextKey = "user"

func AuthMiddleware(jwtManager *auth.JWTManager) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            header := r.Header.Get("Authorization")
            if !strings.HasPrefix(header, "Bearer ") {
                http.Error(w, "missing auth header", http.StatusUnauthorized); return
            }
            claims, err := jwtManager.ValidateToken(strings.TrimPrefix(header, "Bearer "))
            if err != nil { http.Error(w, "invalid token", http.StatusUnauthorized); return }
            ctx := context.WithValue(r.Context(), UserContextKey, claims)
            next.ServeHTTP(w, r.WithContext(ctx))
        })
    }
}

func GetUserClaims(r *http.Request) *auth.Claims {
    c, _ := r.Context().Value(UserContextKey).(*auth.Claims)
    return c
}
```

```go
package handler

import (
    "crypto/sha256"
    "encoding/hex"
    "encoding/json"
    "net/http"
    "time"
    "github.com/google/uuid"
    "github.com/tahsinarafat/aioj/internal/auth"
    "github.com/tahsinarafat/aioj/internal/model"
    "github.com/tahsinarafat/aioj/internal/store"
)

type AuthHandler struct {
    users       store.UserStore
    refreshToks store.RefreshTokenStore
    jwt         *auth.JWTManager
}

func NewAuthHandler(users store.UserStore, refreshToks store.RefreshTokenStore, jwt *auth.JWTManager) *AuthHandler {
    return &AuthHandler{users: users, refreshToks: refreshToks, jwt: jwt}
}

func (h *AuthHandler) Register(w http.ResponseWriter, r *http.Request) {
    var req model.RegisterRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "invalid body", http.StatusBadRequest); return
    }
    if req.Username == "" || req.Email == "" || req.Password == "" {
        http.Error(w, "username, email, password required", http.StatusBadRequest); return
    }
    existing, _ := h.users.GetByUsername(r.Context(), req.Username)
    if existing != nil { http.Error(w, "username taken", http.StatusConflict); return }
    existing, _ = h.users.GetByEmail(r.Context(), req.Email)
    if existing != nil { http.Error(w, "email taken", http.StatusConflict); return }

    hash, _ := auth.HashPassword(req.Password)
    user := &model.User{ID: uuid.New().String(), Username: req.Username, Email: req.Email, PasswordHash: hash, Role: "user"}
    if err := h.users.Create(r.Context(), user); err != nil {
        http.Error(w, "failed to create", http.StatusInternalServerError); return
    }
    respondJSON(w, http.StatusCreated, h.tokenResp(r.Context(), user))
}

func (h *AuthHandler) Login(w http.ResponseWriter, r *http.Request) {
    var req model.LoginRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "invalid body", http.StatusBadRequest); return
    }
    user, err := h.users.GetByUsername(r.Context(), req.Username)
    if err != nil || user == nil || !auth.CheckPassword(req.Password, user.PasswordHash) {
        http.Error(w, "invalid credentials", http.StatusUnauthorized); return
    }
    respondJSON(w, http.StatusOK, h.tokenResp(r.Context(), user))
}

func (h *AuthHandler) Refresh(w http.ResponseWriter, r *http.Request) {
    var req struct{ RefreshToken string `json:"refresh_token"` }
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "invalid body", http.StatusBadRequest); return
    }
    hsh := sha256.Sum256([]byte(req.RefreshToken))
    userID, err := h.refreshToks.Validate(r.Context(), hex.EncodeToString(hsh[:]))
    if err != nil { http.Error(w, "invalid refresh token", http.StatusUnauthorized); return }
    user, _ := h.users.GetByID(r.Context(), userID)
    if user == nil { http.Error(w, "user not found", http.StatusUnauthorized); return }
    respondJSON(w, http.StatusOK, h.tokenResp(r.Context(), user))
}

func (h *AuthHandler) tokenResp(ctx context.Context, user *model.User) *model.AuthResponse {
    access, _ := h.jwt.GenerateAccessToken(user.ID, user.Username, user.Role)
    raw, hashed := h.jwt.GenerateRefreshToken()
    h.refreshToks.Create(ctx, user.ID, hashed, time.Now().Add(h.jwt.RefreshTTL()))
    return &model.AuthResponse{AccessToken: access, RefreshToken: raw, User: user}
}

func respondJSON(w http.ResponseWriter, status int, data interface{}) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(status)
    json.NewEncoder(w).Encode(data)
}
```

```go
    Key                   string            `yaml:"key"`

    TimeLimitMultiplier   float64           `yaml:"time_limit_multiplier"`

    Extensions            []string          `yaml:"extensions"`
    Mono                  bool              `yaml:"mono,omitempty"`
    Runtime               string            `yaml:"runtime,omitempty"`
}

func LoadLanguages(dir string) (map[string]*LangConfig, error) {
    entries, err := os.ReadDir(dir)
    if err != nil { return nil, fmt.Errorf("read lang dir: %w", err) }
    langs := make(map[string]*LangConfig)
    for _, e := range entries {
        if e.IsDir() || filepath.Ext(e.Name()) != ".yaml" { continue }
        data, _ := os.ReadFile(filepath.Join(dir, e.Name()))
        var cfg LangConfig
        if err := yaml.Unmarshal(data, &cfg); err != nil { continue }
        if cfg.Key == "" { continue }
        langs[cfg.Key] = &cfg
    }
    return langs, nil
}
```

```go
// internal/queue/memory.go
package queue

import (
    "context"
    "sync"
)

type MemoryQueue struct {
    mu    sync.Mutex
    cond  *sync.Cond
    queue []string
}

func NewMemory() *MemoryQueue {
    m := &MemoryQueue{}
    m.cond = sync.NewCond(&m.mu)
    return m
}

func (m *MemoryQueue) Enqueue(_ context.Context, id string) error {
    m.mu.Lock()
    m.queue = append(m.queue, id)
    m.cond.Signal()
    m.mu.Unlock()
    return nil
}

func (m *MemoryQueue) Dequeue(ctx context.Context) (string, error) {
    m.mu.Lock()
    defer m.mu.Unlock()
    for len(m.queue) == 0 {
        select {
        case <-ctx.Done(): return "", ctx.Err()
        default:
        }
        m.cond.Wait()
    }
    id := m.queue[0]
    m.queue = m.queue[1:]
    return id, nil
}

func (m *MemoryQueue) Len() int { m.mu.Lock(); defer m.mu.Unlock(); return len(m.queue) }
func (m *MemoryQueue) Close() error { m.cond.Broadcast(); return nil }
```

```go
// internal/judge/checker/checker.go
package checker

import (
    "bytes"
    "math"
    "strconv"
    "strings"
)

type Result struct {
    Passed  bool
    Score   int
    Message string
}

type Checker interface {
    Name() string
    Check(input, expected, actual []byte) *Result
}

type ExactChecker struct{}
func (ExactChecker) Name() string { return "exact" }
func (ExactChecker) Check(_, expected, actual []byte) *Result {
    if bytes.Equal(bytes.TrimSpace(expected), bytes.TrimSpace(actual)) {
        return &Result{Passed: true, Score: 100, Message: "OK"}
    }
    return &Result{Passed: false, Score: 0, Message: "output mismatch"}
}

type LinesChecker struct{}
func (LinesChecker) Name() string { return "lines" }
func (LinesChecker) Check(_, expected, actual []byte) *Result {
    exp := strings.Split(string(bytes.TrimSpace(expected)), "\n")
    act := strings.Split(string(bytes.TrimSpace(actual)), "\n")
    if len(exp) != len(act) {
        return &Result{Passed: false, Score: 0, Message: "line count mismatch"}
    }
    for i := range exp {
        if strings.TrimSpace(exp[i]) != strings.TrimSpace(act[i]) {
            return &Result{Passed: false, Score: 0, Message: "line " + strconv.Itoa(i+1) + " differs"}
        }
    }
    return &Result{Passed: true, Score: 100, Message: "OK"}
}

type FloatChecker struct{ Epsilon float64 }
func (f FloatChecker) Name() string { return "float" }
func (f FloatChecker) Check(_, expected, actual []byte) *Result {
    exp := strings.Fields(string(bytes.TrimSpace(expected)))
    act := strings.Fields(string(bytes.TrimSpace(actual)))
    if len(exp) != len(act) {
        return &Result{Passed: false, Score: 0, Message: "token count mismatch"}
    }
    eps := f.Epsilon
    if eps == 0 { eps = 1e-6 }
    for i := range exp {
        ev, _ := strconv.ParseFloat(exp[i], 64)
        av, _ := strconv.ParseFloat(act[i], 64)
        if math.Abs(ev-av) > eps {
            return &Result{Passed: false, Score: 0, Message: "diff at " + strconv.Itoa(i+1)}
        }
    }
    return &Result{Passed: true, Score: 100, Message: "OK"}
}

func GetChecker(name string) Checker {
    switch name {
    case "lines": return LinesChecker{}
    case "float": return FloatChecker{Epsilon: 1e-6}
    default: return ExactChecker{}
    }
}
```

```go
type WorkerPool struct {
    queue     queue.JudgeQueue
    exec      *executor.Client
    langDir   string
    sem       chan struct{}
    subStore  store.SubmissionStore
    probStore store.ProblemStore
}

func NewWorkerPool(q queue.JudgeQueue, exec *executor.Client, langDir string, concurrency int,
    subStore store.SubmissionStore, probStore store.ProblemStore) *WorkerPool {
    return &WorkerPool{
        queue: q, exec: exec, langDir: langDir, sem: make(chan struct{}, concurrency),
        subStore: subStore, probStore: probStore,
    }
}

func (wp *WorkerPool) Start(ctx context.Context) {
    for {
        subID, err := wp.queue.Dequeue(ctx)
        if err != nil { return }
        select {
        case wp.sem <- struct{}{}:
            go func(id string) { defer func() { <-wp.sem }(); wp.judge(ctx, id) }(subID)
        case <-ctx.Done(): return
        }
    }
}

func (wp *WorkerPool) judge(ctx context.Context, submissionID string) {
    slog.Info("judging", "id", submissionID)
    sub, err := wp.subStore.GetByID(ctx, submissionID)
    if err != nil || sub == nil { slog.Error("load sub failed", "id", submissionID, "error", err); return }

    wp.subStore.UpdateStatus(ctx, submissionID, model.StatusJudging)

    prob, err := wp.probStore.GetByID(ctx, sub.ProblemID)
    if err != nil || prob == nil {
        wp.subStore.UpdateResult(ctx, submissionID, model.StatusSE, 0, 0, 0, "problem not found", nil)
        return
    }

    langs, _ := compiler.LoadLanguages(wp.langDir)
    cfg := langs[sub.Language]
    if cfg == nil {
        wp.subStore.UpdateResult(ctx, submissionID, model.StatusCE, 0, 0, 0, "unsupported language: "+sub.Language, nil)
        return
    }

    // Compile if needed
    var compiledPath string
    compileOutput := ""
    if cfg.CompileCmd != "" {
        srcName := "Main" + cfg.Extensions[0]
        exeName := "Main"
        cmdStr := cfg.CompileCmd
        cmdStr = strings.ReplaceAll(cmdStr, "{{exe}}", exeName)
        cmdStr = strings.ReplaceAll(cmdStr, "{{src}}", srcName)
        cmdStr = strings.ReplaceAll(cmdStr, "{{dir}}", "/box")

        copyIn := map[string]executor.CmdFile{srcName: {Content: sub.SourceCode}}
        for k, v := range cfg.CopyIn {
            copyIn[k] = executor.CmdFile{Src: v}
        }

        slog.Info("compiling", "lang", sub.Language, "cmd", cmdStr)
        resp, err := wp.exec.Run(&executor.ExecRequest{
            Cmd: []executor.Cmd{{
                Args: []string{"/bin/sh", "-c", cmdStr},
                Env:  []string{"PATH=/usr/bin:/bin"},
                CPULimit: 30000000000, MemoryLimit: 536870912, ProcLimit: 64,
                CopyIn: copyIn, CopyOut: []string{exeName},
            }},
        })
        if err != nil || resp.Results[0].Status != "Accepted" {
            ce := "compile error"
            if err != nil { ce = err.Error() }
            wp.subStore.UpdateResult(ctx, submissionID, model.StatusCE, 0, 0, 0, ce, nil)
            return
        }
        compiledPath = filepath.Join(resp.Results[0].RunDir, exeName)
        _ = compiledPath
    }

    // Interpreted language: copy source into sandbox

    // Run test cases

            Cmd: []executor.Cmd{{
                Args: args, Env: []string{"PATH=/usr/bin:/bin"},
                CPULimit: cpuLimit, MemoryLimit: memLimit, ProcLimit: 16,
                CopyIn: copyIn,
            }},
            PipeInput: true,
        })
        elapsed := time.Since(start)

        r := model.TestCaseResult{CaseName: tc.InputName, Time: int(elapsed.Milliseconds())}
        if err != nil {
            r.Status, r.Detail = model.StatusSE, err.Error()
        } else if len(resp.Results) == 0 {
            r.Status, r.Detail = model.StatusSE, "no result"
        } else {
            cr := resp.Results[0]
            r.Memory = int(cr.Memory / 1024)

        if r.Time > maxTime { maxTime = r.Time }
        if r.Memory > maxMem { maxMem = r.Memory }
        if r.Status != model.StatusAC && finalStatus == model.StatusAC {
            finalStatus = r.Status
        }
        totalScore += r.Score
    }

    avgScore := 0
    if len(prob.TestCaseScore) > 0 { avgScore = totalScore / len(prob.TestCaseScore) }

    wp.subStore.UpdateResult(ctx, submissionID, finalStatus, avgScore, maxTime, maxMem, compileOutput, results)
    wp.probStore.UpdateCounts(ctx, prob.ID, 1, boolToInt(finalStatus == model.StatusAC))
    slog.Info("judged", "id", submissionID, "verdict", finalStatus, "time", maxTime, "mem", maxMem)
}

    data, err := os.ReadFile(path)
    if err != nil { return "" }
    return string(data)
}
```

```go
package handler

import (
    "encoding/json"
    "log/slog"
    "net/http"
    "sync"
    "github.com/gorilla/websocket"
)

var upgrader = websocket.Upgrader{CheckOrigin: func(*http.Request) bool { return true }}

type WSMessage struct {
    Type         string      `json:"type"`
    SubmissionID string      `json:"submission_id,omitempty"`
    ContestID    string      `json:"contest_id,omitempty"`
    Data         interface{} `json:"data,omitempty"`
}

type WSManager struct {
    mu       sync.RWMutex
    subs     map[string]map[*websocket.Conn]bool
    contests map[string]map[*websocket.Conn]bool
}

func NewWSManager() *WSManager {
    return &WSManager{subs: make(map[string]map[*websocket.Conn]bool), contests: make(map[string]map[*websocket.Conn]bool)}
}

func (m *WSManager) Handle(w http.ResponseWriter, r *http.Request) {
    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil { slog.Error("ws upgrade", "error", err); return }
    defer conn.Close()
    for {
        var msg WSMessage
        if err := conn.ReadJSON(&msg); err != nil { break }
        switch msg.Type {
        case "subscribe":
            m.mu.Lock()
            if msg.SubmissionID != "" {
                if m.subs[msg.SubmissionID] == nil { m.subs[msg.SubmissionID] = make(map[*websocket.Conn]bool) }
                m.subs[msg.SubmissionID][conn] = true
            }
            if msg.ContestID != "" {
                if m.contests[msg.ContestID] == nil { m.contests[msg.ContestID] = make(map[*websocket.Conn]bool) }
                m.contests[msg.ContestID][conn] = true
            }
            m.mu.Unlock()
        }
    }
}

func (m *WSManager) NotifySubmission(id string, msg WSMessage) {
    m.mu.RLock()
    conns := m.subs[id]
    m.mu.RUnlock()
    data, _ := json.Marshal(msg)
    for c := range conns { c.WriteMessage(websocket.TextMessage, data) }
}

func (m *WSManager) NotifyContest(id string, msg WSMessage) {
    m.mu.RLock()
    conns := m.contests[id]
    m.mu.RUnlock()
    data, _ := json.Marshal(msg)
    for c := range conns { c.WriteMessage(websocket.TextMessage, data) }
}
```

```ts
const BASE = '/api'
let accessToken = localStorage.getItem('access_token')
let refreshToken = localStorage.getItem('refresh_token')

export function setTokens(a: string, r: string) {
  accessToken = a; refreshToken = r
  localStorage.setItem('access_token', a); localStorage.setItem('refresh_token', r)
}
export function clearTokens() {
  accessToken = refreshToken = null
  localStorage.removeItem('access_token'); localStorage.removeItem('refresh_token')
}
export function getAccessToken() { return accessToken }

async function request<T>(path: string, opts: RequestInit = {}): Promise<T> {
  const headers: Record<string,string> = {'Content-Type':'application/json', ...opts.headers as any}
  if (accessToken) headers['Authorization'] = `Bearer ${accessToken}`
  let res = await fetch(BASE + path, { ...opts, headers })
  if (res.status === 401 && refreshToken) {
    const ref = await fetch(BASE + '/auth/refresh', { method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({refresh_token:refreshToken}) })
    if (ref.ok) { const d = await ref.json(); setTokens(d.access_token, d.refresh_token); headers['Authorization'] = `Bearer ${accessToken}`; res = await fetch(BASE + path, { ...opts, headers }) }
    else { clearTokens(); throw new Error('session expired') }
  }
  if (!res.ok) throw new Error(await res.text() || `HTTP ${res.status}`)
  return res.json()
}

export const api = {
  auth: {
    register: (d:any) => request('/auth/register', {method:'POST', body:JSON.stringify(d)}),
```

```go
             ORDER BY s.created_at ASC`, contestID)
```

```go
    }

func (h *ContestHandler) Create(w http.ResponseWriter, r *http.Request) {

        Description string `json:"description"`
        StartTime string `json:"start_time"`
        EndTime string `json:"end_time"`
        FreezeTime *string `json:"freeze_time,omitempty"`
        ScoringRule string `json:"scoring_rule"`
        ProblemIDs []string `json:"problem_ids"`
    }
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "invalid body", http.StatusBadRequest); return
    }
    if req.Title == "" || req.StartTime == "" || req.EndTime == "" {
        http.Error(w, "title, start_time, end_time required", http.StatusBadRequest); return
    }
    st, _ := time.Parse(time.RFC3339, req.StartTime)
    et, _ := time.Parse(time.RFC3339, req.EndTime)
    if st.IsZero() || et.IsZero() { http.Error(w, "invalid time format, use RFC3339", http.StatusBadRequest); return }
    if req.ScoringRule == "" { req.ScoringRule = "acm" }

    status := "upcoming"
    now := time.Now()
    if now.After(st) && now.Before(et) { status = "running" } else if now.After(et) { status = "ended" }

    c := &model.Contest{
        ID: uuid.New().String(), Title: req.Title, Description: req.Description,
        StartTime: st, EndTime: et, ScoringRule: req.ScoringRule, CreatedBy: claims.UserID,
        Status: status,
    }
    if req.FreezeTime != nil {
        ft, _ := time.Parse(time.RFC3339, *req.FreezeTime)
        if !ft.IsZero() { c.FreezeTime = &ft }
    }
    if err := h.store.Create(r.Context(), c); err != nil {
        http.Error(w, "create failed", http.StatusInternalServerError); return
    }
    for i, pid := range req.ProblemIDs {
        h.store.AddProblem(r.Context(), model.ContestProblem{
            ContestID: c.ID, ProblemID: pid, Ordinal: i, Label: string(rune('A'+i)),
        })
    }
    respondJSON(w, http.StatusCreated, c)
```

```go
package handler

import (
    "net/http"
    "sort"
    "github.com/go-chi/chi/v5"
    "github.com/tahsinarafat/aioj/internal/model"
)

type ScoreboardHandler struct {
    contestStore store.ContestStore
    userStore    store.UserStore
}

func NewScoreboardHandler(cs store.ContestStore, us store.UserStore) *ScoreboardHandler {
    return &ScoreboardHandler{contestStore: cs, userStore: us}
}

func (h *ScoreboardHandler) GetScoreboard(w http.ResponseWriter, r *http.Request) {
    contestID := chi.URLParam(r, "id")
    contest, _ := h.contestStore.GetByID(r.Context(), contestID)
    if contest == nil { http.Error(w, "contest not found", http.StatusNotFound); return }

    problems, _ := h.contestStore.GetContestProblems(r.Context(), contestID)
    userIDs, _ := h.contestStore.GetRegisteredUserIDs(r.Context(), contestID)

    now := time.Now()
    frozen := contest.FreezeTime != nil && now.After(*contest.FreezeTime) && now.Before(contest.EndTime)
    subs, _ := h.contestStore.GetSubmissionsForScoreboard(r.Context(), contestID, frozen)
    startTime := contest.StartTime

    // Build per-user, per-problem result map
    type userProbKey struct{ userID, problemID string }
    best := make(map[userProbKey]*model.ProblemResult)

    for _, sub := range subs {
        key := userProbKey{sub.UserID, sub.ProblemID}
        pr, exists := best[key]
        if !exists { pr = &model.ProblemResult{}; best[key] = pr }

        minutes := int(sub.CreatedAt.Sub(startTime).Minutes())
        pr.Time = minutes

        if contest.ScoringRule == "acm" {
            if sub.Status == model.StatusAccepted {
                if !pr.Solved {
                    pr.Solved = true
                    pr.Penalty = pr.Attempts * 20 + minutes
                }
            } else if sub.Status == model.StatusWrongAnswer || sub.Status == model.StatusRuntimeError || sub.Status == model.StatusTimeLimitExceeded {
                if !pr.Solved { pr.Attempts++ }
            }
        } else if contest.ScoringRule == "oi" {
            if sub.Score > pr.Score {
                pr.Score = sub.Score
                pr.Solved = sub.Score >= 99.9
            }
        }
    }

    // Build entries
    entriesMap := make(map[string]*model.ScoreboardEntry)
    for _, uid := range userIDs {
        username, _ := h.userStore.GetUsername(r.Context(), uid)
        entriesMap[uid] = &model.ScoreboardEntry{
            UserID: uid, Username: username,
            Problems: make(map[string]model.ProblemResult),
        }
    }
    for key, pr := range best {
        entry := entriesMap[key.userID]
        if entry == nil { continue }
        // Find label
        label := ""
        for _, p := range problems {
            if p.ProblemID == key.problemID { label = p.Label; break }
        }
        if frozen && label != "" {
            entry.Problems[label] = model.ProblemResult{Attempts: pr.Attempts, Solved: false, Score: 0, Penalty: 0}
            // In freeze, show last known AC as "?" — we show pending state
        } else {
            entry.Problems[label] = *pr
        }
        if contest.ScoringRule == "acm" {
            if pr.Solved { entry.TotalSolved++; entry.TotalPenalty += pr.Penalty }
        } else {
            entry.TotalScore += pr.Score
        }
    }

    entries := make([]*model.ScoreboardEntry, 0, len(entriesMap))
    for _, e := range entriesMap { entries = append(entries, e) }

    if contest.ScoringRule == "acm" {
        sort.Slice(entries, func(i, j int) bool {
            if entries[i].TotalSolved != entries[j].TotalSolved { return entries[i].TotalSolved > entries[j].TotalSolved }
            return entries[i].TotalPenalty < entries[j].TotalPenalty
        })
    } else {
        sort.Slice(entries, func(i, j int) bool { return entries[i].TotalScore > entries[j].TotalScore })
    }
    for i := range entries { entries[i].Rank = i + 1 }

    respondJSON(w, http.StatusOK, map[string]interface{}{
        "entries": entries, "frozen": frozen, "problems": problems,
    })
}
```

```go
package vjudge

import (
    "context"
    "fmt"
    "log/slog"
    "sync"
    "time"
    "encoding/json"
    "github.com/tahsinarafat/aioj/internal/model"
    "github.com/tahsinarafat/aioj/internal/store"
)

type VJSubmissionRequest struct {
    ID                   string `json:"id"`
    ProblemRemoteID      string `json:"problem_remote_id"`
    SourceCode           string `json:"source_code"`
    Language             string `json:"language"`
    RemoteOJ             string `json:"remote_oj"`
}

type VJResult struct {
    ID         string  `json:"id"`
    Verdict    string  `json:"verdict"`
    TimeUsed   int     `json:"time_used_ms"`
    MemoryUsed int     `json:"memory_used_kb"`
    Score      float64 `json:"score"`
}

type Service struct {
    bots     map[string]Bot
    mu       sync.RWMutex
    subStore store.SubmissionStore
    ws       *handler.WSManager
}

func NewService(subStore store.SubmissionStore, ws *handler.WSManager) *Service {
    return &Service{
        bots:     make(map[string]Bot),
        subStore: subStore,
        ws:       ws,
    }
}

func (s *Service) RegisterBot(name string, bot Bot) {
    s.mu.Lock()
    defer s.mu.Unlock()
    s.bots[name] = bot
    slog.Info("registered VJudge bot", "name", name)
}

func (s *Service) Submit(ctx context.Context, req VJSubmissionRequest) error {
    s.mu.RLock()
    bot, ok := s.bots[req.RemoteOJ]
    s.mu.RUnlock()
    if !ok {
        return fmt.Errorf("no bot registered for remote OJ: %s", req.RemoteOJ)
    }

    remoteID, err := bot.Submit(ctx, req.ProblemRemoteID, req.SourceCode, req.Language)
    if err != nil {
        s.subStore.UpdateStatus(ctx, req.ID, model.StatusJudgeError)
        return fmt.Errorf("submit to %s failed: %w", req.RemoteOJ, err)
    }

    // Poll in background
    go s.pollLoop(req.ID, remoteID, bot)
    return nil
}

func (s *Service) pollLoop(submissionID, remoteID string, bot Bot) {
    ctx := context.Background()
    ticker := time.NewTicker(3 * time.Second)
    defer ticker.Stop()
    timeout := time.After(5 * time.Minute)

    for {
        select {
        case <-ticker.C:
            result, err := bot.Poll(ctx, remoteID)
            if err != nil {
                slog.Error("vjudge poll error", "submission", submissionID, "error", err)
                continue
            }
            if result == nil || !result.Done { continue }

            // Map verdict
            var status model.SubmissionStatus
            var score float64
            switch result.Verdict {
            case "AC", "Accepted":
                status = model.StatusAccepted; score = 100
            case "WA", "Wrong answer":
                status = model.StatusWrongAnswer
            case "TLE", "Time limit exceeded":
                status = model.StatusTimeLimitExceeded
            case "CE", "Compilation error":
                status = model.StatusCompileError
            case "RE", "Runtime error":
                status = model.StatusRuntimeError
            default:
                status = model.StatusWrongAnswer
            }

            s.subStore.UpdateResult(ctx, submissionID, status, int(score), result.TimeUsed, result.MemoryUsed, "", nil)

            if s.ws != nil {
                s.ws.NotifySubmission(submissionID, handler.WSMessage{
                    Type: "judge_result",
                    Data: map[string]interface{}{
                        "submission_id": submissionID,
                        "status": status,
                        "score": score,
                        "time_used": result.TimeUsed,
                        "memory_used": result.MemoryUsed,
                        "verdict": result.Verdict,
                    },
                })
            }
            return

        case <-timeout:
            slog.Warn("vjudge poll timeout", "submission", submissionID)
            return
        }
    }
}

func (s *Service) GetBots() []string {
    s.mu.RLock()
    defer s.mu.RUnlock()
    names := make([]string, 0, len(s.bots))
    for name := range s.bots { names = append(names, name) }
    return names
}
```

```go
package vjudge

import (
    "context"
    "fmt"
    "net/http"
    "net/url"
    "strings"
    "time"
    "golang.org/x/net/html"
    "io"
)

type CodeforcesBot struct {
    config  BotConfig
    client  *http.Client
    cookies []*http.Cookie
    state   BotState
}

func NewCodeforcesBot(cfg BotConfig) *CodeforcesBot {
    return &CodeforcesBot{
        config: cfg,
        client: &http.Client{Timeout: 30 * time.Second, Transport: &http.Transport{DisableKeepAlives: false}},
        state:  StateIdle,
    }
}

func (b *CodeforcesBot) Name() string { return "codeforces" }

func (b *CodeforcesBot) State() BotState { return b.state }

func (b *CodeforcesBot) login(ctx context.Context) error {
    // Get CSRF token
    req, _ := http.NewRequestWithContext(ctx, "GET", "https://codeforces.com/enter", nil)
    resp, err := b.client.Do(req)
    if err != nil { return err }
    defer resp.Body.Close()
    b.cookies = resp.Cookies()
    csrf := extractCSRF(resp.Body)

    // Post login form
    data := url.Values{
        "csrf_token": {csrf}, "action": {"enter"},
        "handleOrEmail": {b.config.Username}, "password": {b.config.Password},
        "remember": {"on"}, "_tta": {"1"},
    }
    loginReq, _ := http.NewRequestWithContext(ctx, "POST", "https://codeforces.com/enter", strings.NewReader(data.Encode()))
    loginReq.Header.Set("Content-Type", "application/x-www-form-urlencoded")
    for _, c := range b.cookies { loginReq.AddCookie(c) }
    loginResp, err := b.client.Do(loginReq)
    if err != nil { return err }
    defer loginResp.Body.Close()
    b.cookies = loginResp.Cookies()

    // Check if login succeeded by looking for "logout" in response
    body, _ := io.ReadAll(loginResp.Body)
    if !strings.Contains(string(body), "logout") {
        // Try alternate page; check for handle in title
        if strings.Contains(string(body), "Invalid handle or password") {
            return fmt.Errorf("codeforces login failed: invalid credentials")
        }
    }
    b.state = StateIdle
    return nil
}

func extractCSRF(body io.Reader) string {
    token := ""
    z := html.NewTokenizer(body)
    for {
        tt := z.Next()
        if tt == html.ErrorToken { break }
        if tt == html.StartTagToken || tt == html.SelfClosingTagToken {
            name, hasAttr := z.TagName()
            if string(name) == "meta" && hasAttr {
                var attrName, attrVal string
                for hasAttr {
                    attrName, attrVal, hasAttr = z.TagAttr()
                    if string(attrName) == "name" && string(attrVal) == "X-Csrf-Token" {
                        // Next attribute is content
                        attrName, attrVal, _ = z.TagAttr()
                        token = string(attrVal)
                        return token
                    }
                }
            }
        }
    }
    return token
}

func (b *CodeforcesBot) Submit(ctx context.Context, problemRemoteID, sourceCode, language string) (string, error) {
    b.state = StateRunning
    if err := b.login(ctx); err != nil {
        b.state = StateError
        return "", err
    }

    // Determine submit page CSRF
    submitURL := fmt.Sprintf("https://codeforces.com/problemset/submit/%s", problemRemoteID)
    req, _ := http.NewRequestWithContext(ctx, "GET", submitURL, nil)
    for _, c := range b.cookies { req.AddCookie(c) }
    resp, err := b.client.Do(req)
    if err != nil { b.state = StateError; return "", err }
    defer resp.Body.Close()
    csrf := extractCSRF(resp.Body)

    // Map language
    langMap := map[string]string{
        "cpp":     "54", // GNU G++17 7.3.0
        "c":       "43", // GNU GCC C11
        "python":  "70", // Python 3.8
        "java":    "60", // Java 11
        "rust":    "75", // Rust 1.42
        "javascript": "34", // Node.js 12
        "csharp":  "67", // C# Mono 6.8
    }
    langID, ok := langMap[language]
    if !ok { langID = "54" }

    data := url.Values{
        "csrf_token": {csrf}, "action": {"submitSolutionFormSubmitted"},
        "submittedProblemCode": {fmt.Sprintf("%s%s", problemRemoteID, "?locale=en")},
        "programTypeId": {langID}, "source": {sourceCode},
        "sourceCodeConfirmed": {"true"}, "tabSize": {"4"}, "script": {"true"},
    }
    // CF expects the problem code as just the number for submit
    data.Set("submittedProblemCode", problemRemoteID)
    // Actually the problem code on submit page
    data.Set("submittedProblemCode", problemRemoteID)

    subReq, _ := http.NewRequestWithContext(ctx, "POST", submitURL, strings.NewReader(data.Encode()))
    subReq.Header.Set("Content-Type", "application/x-www-form-urlencoded")
    subReq.Header.Set("User-Agent", "Mozilla/5.0 (compatible; AIOJ/1.0)")
    for _, c := range b.cookies { subReq.AddCookie(c) }
    subResp, err := b.client.Do(subReq)
    if err != nil { b.state = StateError; return "", err }
    defer subResp.Body.Close()

    // Extract submission ID from redirect URL or response
    finalURL := subResp.Request.URL.String()
    b.state = StateIdle

    // Parse submission ID from URL like /problemset/submission/1234/567890
    parts := strings.Split(finalURL, "/")
    if len(parts) >= 2 {
        return parts[len(parts)-1], nil
    }
    return "", fmt.Errorf("could not extract submission ID from URL: %s", finalURL)
}

func (b *CodeforcesBot) Poll(ctx context.Context, remoteSubmissionID string) (*RemoteSubmissionResult, error) {
    // Get submission status page
    url := fmt.Sprintf("https://codeforces.com/problemset/submission?submissionId=%s", remoteSubmissionID)
    req, _ := http.NewRequestWithContext(ctx, "GET", url, nil)
    for _, c := range b.cookies { req.AddCookie(c) }
    resp, err := b.client.Do(req)
    if err != nil { return nil, err }
    defer resp.Body.Close()

    // Parse HTML for verdict
    // CF shows verdict in a <span class="verdict-*"> or <span submissionVerdict="AC">
    body, _ := io.ReadAll(resp.Body)
    content := string(body)

    result := &RemoteSubmissionResult{RemoteID: remoteSubmissionID, Done: false}

    // Look for verdict patterns
    if strings.Contains(content, "verdict-accepted") || strings.Contains(content, "submissionVerdict=\"OK\"") {
        result.Verdict = "AC"; result.Done = true; result.Status = "Accepted"
    } else if strings.Contains(content, "verdict-wrong-answer") || strings.Contains(content, "submissionVerdict=\"WRONG_ANSWER\"") {
        result.Verdict = "WA"; result.Done = true; result.Status = "Wrong answer"
    } else if strings.Contains(content, "verdict-time-limit-exceeded") || strings.Contains(content, "submissionVerdict=\"TIME_LIMIT_EXCEEDED\"") {
        result.Verdict = "TLE"; result.Done = true; result.Status = "Time limit exceeded"
    } else if strings.Contains(content, "verdict-compilation-error") || strings.Contains(content, "submissionVerdict=\"COMPILATION_ERROR\"") {
        result.Verdict = "CE"; result.Done = true; result.Status = "Compilation error"
    } else if strings.Contains(content, "verdict-runtime-error") || strings.Contains(content, "submissionVerdict=\"RUNTIME_ERROR\"") {
        result.Verdict = "RE"; result.Done = true; result.Status = "Runtime error"
    } else if strings.Contains(content, "verdict-running") || strings.Contains(content, "In queue") || strings.Contains(content, "Running") {
        result.Done = false; result.Status = "Pending"
    } else {
        // Check if submission page is valid
        if strings.Contains(content, "submissionId") || strings.Contains(content, "submissionVerdict") {
            result.Done = false; result.Status = "Unknown"
        } else {
            // Might be invalid or still processing
            return nil, nil
        }
    }
    return result, nil
}
```

```go
func (h *VJudgeHandler) Submit(w http.ResponseWriter, r *http.Request) {
    var req vjudge.VJSubmissionRequest

        http.Error(w, "invalid body", http.StatusBadRequest); return

        http.Error(w, err.Error(), http.StatusBadRequest); return
    }
    w.WriteHeader(http.StatusAccepted)
}
```
